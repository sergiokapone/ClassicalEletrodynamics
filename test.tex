\begin{problem}%
Показати, що слід тензора $T^{\mu\nu}$ дорівнює нулю:
\begin{equation*}
    T^{\mu}{}_{\mu} = 0.
\end{equation*}
\end{problem}


%=========================================================
\begin{problem}%
Обчислити згортку $T^{\mu\nu}T_{\mu\nu}$ та виразити її через інваріанти
$\mathcal{I}_1$ та $\mathcal{I}_2$:
\begin{equation*}
    T^{\mu\nu}T_{\mu\nu} = \frac{2}{(4\pi)^2} (\mathcal{I}_1^2 + 4\mathcal{I}_2^2),
\end{equation*}
де
\begin{equation*}
    \mathcal{I}_1 = \frac{1}{2}F_{\mu\nu}F^{\mu\nu} = B^2 - E^2,
    \qquad
    \mathcal{I}_2 = \frac{1}{4}F_{\mu\nu}\tilde{F}^{\mu\nu} = \vec{E}\cdot\vec{B},
\end{equation*}
де $\tilde{F}^{\mu\nu} = \frac{1}{2}\varepsilon^{\mu\nu\lambda\sigma}F_{\lambda\sigma}$ --- дуальний тензор.
\end{problem}

%=========================================================
\begin{problem}%
Використовуючи явні вирази для компонент $T^{\mu\nu}$ через
густину енергії $w$ та вектор Пойнтінга $\vec{S}$,
\begin{equation*}
    T^{00} = w = \frac{E^2 + B^2}{8\pi},
    \qquad
    T^{0i} = \frac{S^i}{c} = \frac{(\vec{E}\times\vec{B})^i}{4\pi c},
\end{equation*}
показати, що
\begin{equation*}
    w^2 - \frac{S^2}{c^2} = \frac{1}{2}\,T^{\mu\nu}T_{\mu\nu}
    = \frac{\mathcal{I}_1^2 + 4\mathcal{I}_2^2}{(4\pi)^2}.
\end{equation*}
За яких умов на поле $w^2 - S^2/c^2 = 0$?
%Який фізичний стан поля відповідає цій умові?
\end{problem}
